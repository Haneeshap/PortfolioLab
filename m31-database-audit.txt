===== V11__scheduled_dividends.sql =====
CREATE TABLE simulated_dividend_schedules (
    id BIGSERIAL PRIMARY KEY,
    asset_id BIGINT NOT NULL REFERENCES assets(id),
    reference_key VARCHAR(120) NOT NULL UNIQUE,
    dividend_per_share NUMERIC(19,4) NOT NULL CHECK (dividend_per_share > 0),
    currency VARCHAR(3) NOT NULL,
    record_cutoff TIMESTAMPTZ NOT NULL,
    payment_at TIMESTAMPTZ NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    CHECK (payment_at > record_cutoff)
);

CREATE TABLE simulated_scheduled_dividend_payments (
    id BIGSERIAL PRIMARY KEY,
    schedule_id BIGINT NOT NULL REFERENCES simulated_dividend_schedules(id),
    portfolio_id BIGINT NOT NULL REFERENCES portfolios(id),
    eligible_shares NUMERIC(19,6) NOT NULL CHECK (eligible_shares > 0),
    amount NUMERIC(19,4) NOT NULL CHECK (amount > 0),
    credited_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    UNIQUE (schedule_id, portfolio_id)
);
CREATE INDEX idx_simulated_dividend_schedules_due ON simulated_dividend_schedules(payment_at);

===== V12__market_price_alerts.sql =====
CREATE TABLE market_price_alerts (
 id BIGSERIAL PRIMARY KEY,
 portfolio_id BIGINT NOT NULL REFERENCES portfolios(id),
 symbol VARCHAR(20) NOT NULL,
 currency VARCHAR(3) NOT NULL,
 direction VARCHAR(5) NOT NULL CHECK (direction IN ('ABOVE','BELOW')),
 target_price NUMERIC(19,4) NOT NULL CHECK (target_price > 0),
 last_price NUMERIC(19,4) NOT NULL CHECK (last_price > 0),
 status VARCHAR(12) NOT NULL DEFAULT 'ACTIVE' CHECK (status IN ('ACTIVE','TRIGGERED','CANCELLED')),
 created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
 triggered_at TIMESTAMPTZ,
 CHECK ((status = 'TRIGGERED') = (triggered_at IS NOT NULL))
);
CREATE INDEX idx_market_price_alerts_active ON market_price_alerts(status, symbol);
CREATE INDEX idx_market_price_alerts_portfolio ON market_price_alerts(portfolio_id, created_at DESC);

-- Simulated market observations and durable, deduplicated intelligence events.
CREATE TABLE market_price_samples (
 id BIGSERIAL PRIMARY KEY, symbol VARCHAR(20) NOT NULL,
 currency VARCHAR(3) NOT NULL, price NUMERIC(19,4) NOT NULL CHECK(price>0),

 id BIGSERIAL PRIMARY KEY, symbol VARCHAR(20) NOT NULL,
 currency VARCHAR(3) NOT NULL, event_type VARCHAR(30) NOT NULL,
 window_seconds INTEGER NOT NULL, start_price NUMERIC(19,4) NOT NULL,
 end_price NUMERIC(19,4) NOT NULL, change_percent NUMERIC(12,4) NOT NULL,

CREATE INDEX market_intelligence_events_latest ON market_intelligence_events(created_at DESC,id DESC);
CREATE INDEX market_intelligence_events_cooldown ON market_intelligence_events(symbol,event_type,created_at DESC);
CREATE TABLE market_intelligence_outbox (
 event_id BIGSERIAL PRIMARY KEY,
 intelligence_id BIGINT NOT NULL UNIQUE REFERENCES market_intelligence_events(id) ON DELETE CASCADE,
 payload TEXT NOT NULL,
 created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
 published_at TIMESTAMPTZ
);
CREATE INDEX market_intelligence_outbox_pending ON market_intelligence_outbox(created_at,event_id) WHERE published_at IS NULL;

===== V3__create_portfolio_tables.sql =====
CREATE TABLE portfolios (
    id BIGSERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    cash_balance NUMERIC(19, 4) NOT NULL DEFAULT 0,
    currency VARCHAR(3) NOT NULL DEFAULT 'INR',
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT chk_portfolio_cash_nonnegative
        CHECK (cash_balance >= 0),

    CONSTRAINT chk_portfolio_currency
        CHECK (currency ~ '^[A-Z]{3}$')
);

CREATE TABLE holdings (
    id BIGSERIAL PRIMARY KEY,
    portfolio_id BIGINT NOT NULL,
    asset_id BIGINT NOT NULL,
    quantity NUMERIC(19, 6) NOT NULL,
    average_buy_price NUMERIC(19, 4) NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_holdings_portfolio
        FOREIGN KEY (portfolio_id)
        REFERENCES portfolios(id),

    CONSTRAINT fk_holdings_asset
        FOREIGN KEY (asset_id)
        REFERENCES assets(id),

    CONSTRAINT uq_portfolio_asset
        UNIQUE (portfolio_id, asset_id),

    CONSTRAINT chk_holding_quantity
        CHECK (quantity > 0),

    CONSTRAINT chk_holding_average_price
        CHECK (average_buy_price >= 0)
);

CREATE TABLE transactions (
    id BIGSERIAL PRIMARY KEY,
    portfolio_id BIGINT NOT NULL,
    asset_id BIGINT NOT NULL,
    transaction_type VARCHAR(4) NOT NULL,
    quantity NUMERIC(19, 6) NOT NULL,
    price_per_unit NUMERIC(19, 4) NOT NULL,
    total_amount NUMERIC(19, 4) NOT NULL,
    currency VARCHAR(3) NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_transactions_portfolio
        FOREIGN KEY (portfolio_id)
        REFERENCES portfolios(id),

    CONSTRAINT fk_transactions_asset
        FOREIGN KEY (asset_id)
        REFERENCES assets(id),

    CONSTRAINT chk_transaction_type
        CHECK (transaction_type IN ('BUY', 'SELL')),

    CONSTRAINT chk_transaction_quantity
        CHECK (quantity > 0),

    CONSTRAINT chk_transaction_price
        CHECK (price_per_unit > 0),

    CONSTRAINT chk_transaction_total
        CHECK (total_amount > 0),

    CONSTRAINT chk_transaction_currency
        CHECK (currency ~ '^[A-Z]{3}$')
);

CREATE INDEX idx_holdings_portfolio
    ON holdings(portfolio_id);

CREATE INDEX idx_transactions_portfolio_created
    ON transactions(portfolio_id, created_at DESC);

===== V8__add_holding_historical_cost.sql =====
ALTER TABLE holdings
    ADD COLUMN historical_cost_portfolio NUMERIC(19,4);

DO $$
DECLARE
    h RECORD;
    t RECORD;
    running_quantity NUMERIC;
    running_cost NUMERIC;
BEGIN
    FOR h IN
        SELECT id, portfolio_id, asset_id, quantity
        FROM holdings
        ORDER BY id
    LOOP
        running_quantity := 0;
        running_cost := 0;

        FOR t IN
            SELECT transaction_type, quantity, portfolio_amount
            FROM transactions
            WHERE portfolio_id = h.portfolio_id
              AND asset_id = h.asset_id
            ORDER BY created_at ASC, id ASC
        LOOP
            IF t.transaction_type = 'BUY' THEN
                running_quantity := running_quantity + t.quantity;
                running_cost := running_cost + t.portfolio_amount;

            ELSIF t.transaction_type = 'SELL' THEN
                IF t.quantity > running_quantity THEN
                    RAISE EXCEPTION
                        'SELL exceeds position for holding %', h.id;
                END IF;

                IF t.quantity = running_quantity THEN
                    running_quantity := 0;
                    running_cost := 0;
                ELSE
                    running_cost := running_cost *
                        (running_quantity - t.quantity) /
                        running_quantity;

                    running_quantity := running_quantity - t.quantity;
                END IF;
            ELSE
                RAISE EXCEPTION
                    'Unknown transaction type for holding %', h.id;
            END IF;
        END LOOP;

        IF running_quantity <> h.quantity THEN
            RAISE EXCEPTION
                'History quantity mismatch for holding %', h.id;
        END IF;

        IF running_quantity <= 0 THEN
            RAISE EXCEPTION
                'Invalid remaining quantity for holding %', h.id;
        END IF;

        UPDATE holdings
        SET historical_cost_portfolio = ROUND(running_cost, 4)
        WHERE id = h.id;
    END LOOP;
END $$;

ALTER TABLE holdings
    ALTER COLUMN historical_cost_portfolio SET NOT NULL;

ALTER TABLE holdings
    ADD CONSTRAINT chk_holding_historical_cost_nonnegative
    CHECK (historical_cost_portfolio >= 0);

===== V9__create_notification_infrastructure.sql =====
CREATE TABLE notification_outbox (
    event_id UUID PRIMARY KEY,
    event_type VARCHAR(80) NOT NULL,
    portfolio_id BIGINT NOT NULL REFERENCES portfolios(id),
    payload TEXT NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    published_at TIMESTAMPTZ
);

CREATE INDEX idx_notification_outbox_pending
    ON notification_outbox (created_at, event_id)
    WHERE published_at IS NULL;

CREATE TABLE notifications (
    id BIGSERIAL PRIMARY KEY,
    event_id UUID NOT NULL UNIQUE,
    portfolio_id BIGINT NOT NULL REFERENCES portfolios(id),
    source VARCHAR(80) NOT NULL,
    title VARCHAR(200) NOT NULL,
    message TEXT NOT NULL,
    target_url VARCHAR(300),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    read_at TIMESTAMPTZ
);

CREATE INDEX idx_notifications_portfolio_created
    ON notifications (portfolio_id, created_at DESC, id DESC);
