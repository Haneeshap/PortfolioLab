package com.portfoliolab.backend.history;

import com.portfoliolab.backend.portfolio.PortfolioRepository;
import org.springframework.http.HttpStatus;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;
import org.springframework.web.server.ResponseStatusException;

import java.util.List;
import java.util.Map;

@RestController
@RequestMapping("/api/portfolios")
public class PortfolioHistoryController {
    private final JdbcTemplate jdbc;
    private final PortfolioRepository portfolios;

    public PortfolioHistoryController(
            JdbcTemplate jdbc,
            PortfolioRepository portfolios
    ) {
        this.jdbc = jdbc;
        this.portfolios = portfolios;
    }

    @GetMapping("/{id}/history")
    public List<Map<String, Object>> history(@PathVariable Long id) {
        if (!portfolios.existsById(id)) {
            throw new ResponseStatusException(
                    HttpStatus.NOT_FOUND, "Portfolio not found"
            );
        }

        return jdbc.queryForList("""
                SELECT captured_at AS "capturedAt",
                       total_value AS "totalValue",
                       cash_balance AS "cashBalance",
                       holdings_value AS "holdingsValue",
                       currency,
                       valuation_source AS "valuationSource"
                FROM (
                    SELECT id, captured_at, total_value,
                           cash_balance, holdings_value,
                           currency, valuation_source
                    FROM portfolio_snapshots
                    WHERE portfolio_id = ?
                    ORDER BY captured_at DESC, id DESC
                    LIMIT 240
                ) recent
                ORDER BY captured_at ASC, id ASC
                """, id);
    }
}

package com.portfoliolab.backend.history;

import com.portfoliolab.backend.market.MarketPrice;
import com.portfoliolab.backend.market.MarketService;
import com.portfoliolab.backend.portfolio.Holding;
import com.portfoliolab.backend.portfolio.HoldingRepository;
import com.portfoliolab.backend.portfolio.Portfolio;
import com.portfoliolab.backend.portfolio.PortfolioRepository;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Service;

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.List;
import java.util.Map;
import java.util.function.Function;
import java.util.stream.Collectors;

@Service
public class PortfolioHistoryRecorder {
    private static final Logger log =
            LoggerFactory.getLogger(PortfolioHistoryRecorder.class);

    private final PortfolioRepository portfolios;
    private final HoldingRepository holdings;
    private final MarketService market;
    private final JdbcTemplate jdbc;

    public PortfolioHistoryRecorder(
            PortfolioRepository portfolios,
            HoldingRepository holdings,
            MarketService market,
            JdbcTemplate jdbc
    ) {
        this.portfolios = portfolios;
        this.holdings = holdings;
        this.market = market;
        this.jdbc = jdbc;
    }

    @Scheduled(fixedDelay = 30000, initialDelay = 5000)
    public void record() {
        List<Portfolio> allPortfolios = portfolios.findAll();

        if (allPortfolios.isEmpty()) {
            return;
        }

        Map<String, MarketPrice> quotes;
        try {
            quotes = market.getCurrentPrices().stream()
                    .collect(Collectors.toMap(
                            MarketPrice::symbol,
                            Function.identity(),
                            (first, second) -> second
                    ));
        } catch (Exception exception) {
            log.warn("Skipping history snapshot: market unavailable");
            return;
        }

        for (Portfolio portfolio : allPortfolios) {
            try {
                recordPortfolio(portfolio, quotes);
            } catch (Exception exception) {
                log.warn(
                        "Could not record portfolio {}: {}",
                        portfolio.getId(),
                        exception.getMessage()
                );
            }
        }
    }

    private void recordPortfolio(
            Portfolio portfolio,
            Map<String, MarketPrice> quotes
    ) {
        List<Holding> positions =
                holdings.findByPortfolioId(portfolio.getId());

        BigDecimal holdingsValue = BigDecimal.ZERO;

        for (Holding holding : positions) {
            String symbol = holding.getAsset().getSymbol();
            String currency = holding.getAsset().getCurrency();

            MarketPrice quote = quotes.get(symbol);

            if (quote == null
                    || quote.price() == null
                    || quote.price().signum() <= 0
                    || !currency.equals(quote.currency())
                    || !portfolio.getCurrency().equals(currency)) {
                log.warn(
                        "Skipping snapshot for portfolio {}: invalid or missing quote for {}",
                        portfolio.getId(),
                        symbol
                );
                return;
            }

            holdingsValue = holdingsValue.add(
                    holding.getQuantity().multiply(quote.price())
            );
        }

        BigDecimal cash = portfolio.getCashBalance();
        BigDecimal total = cash.add(holdingsValue);

        jdbc.update("""
                INSERT INTO portfolio_snapshots
                (portfolio_id, cash_balance, holdings_value,
                 total_value, currency, valuation_source)
                VALUES (?, ?, ?, ?, ?, ?)
                """,
                portfolio.getId(),
                cash,
                holdingsValue.setScale(4, RoundingMode.HALF_UP),
                total.setScale(4, RoundingMode.HALF_UP),
                portfolio.getCurrency(),
                "GO_SIMULATOR"
        );
    }
}

import java.util.Map;
import java.util.UUID;

@Component
public class PortfolioEventOutbox {
    private final JdbcTemplate jdbc;
    private final ObjectMapper mapper;
    public PortfolioEventOutbox(JdbcTemplate jdbc, ObjectMapper mapper) {
        this.jdbc = jdbc;
        this.mapper = mapper;
    }
    public void record(long portfolioId, String type, Map<String, Object> attributes) {
        try {
            UUID id = UUID.randomUUID();
            var event = new java.util.HashMap<String, Object>(attributes);
            event.put("eventId", id.toString());
            event.put("eventType", type);
            event.put("portfolioId", portfolioId);
            jdbc.update("INSERT INTO notification_outbox (event_id, event_type, portfolio_id, payload) VALUES (?, ?, ?, ?)",
                    id, type, portfolioId, mapper.writeValueAsString(event));
        } catch (Exception e) {
            throw new IllegalStateException("Unable to record portfolio event", e);
        }
    }
}

package com.portfoliolab.backend.portfolio;

import com.portfoliolab.backend.fx.FrankfurterRateService;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.server.ResponseStatusException;

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.List;

@Service
public class PortfolioQueryService {

    private final PortfolioRepository portfolioRepository;
    private final HoldingRepository holdingRepository;
    private final TransactionRepository transactionRepository;
    private final FrankfurterRateService frankfurterRateService;

    public PortfolioQueryService(
            PortfolioRepository portfolioRepository,
            HoldingRepository holdingRepository,
            TransactionRepository transactionRepository,
            FrankfurterRateService frankfurterRateService
    ) {
        this.portfolioRepository = portfolioRepository;
        this.holdingRepository = holdingRepository;
        this.transactionRepository = transactionRepository;
        this.frankfurterRateService = frankfurterRateService;
    }

    @Transactional(readOnly = true)
    public List<HoldingResponse> getHoldings(Long portfolioId) {
        Portfolio portfolio = portfolioRepository.findById(portfolioId)
                .orElseThrow(() -> new ResponseStatusException(
                        HttpStatus.NOT_FOUND,
                        "Portfolio not found"
                ));

        String portfolioCurrency = portfolio.getCurrency();

        return holdingRepository.findByPortfolioId(portfolioId)
                .stream()
                .map(holding -> {
                    var asset = holding.getAsset();

                    BigDecimal quantity = holding.getQuantity();
                    BigDecimal currentPrice = asset.getPrice();
                    BigDecimal averagePrice = holding.getAverageBuyPrice();

                    BigDecimal marketValue =
                            quantity.multiply(currentPrice);

                    BigDecimal costBasis =
                            quantity.multiply(averagePrice);

                    BigDecimal profitLoss =
                            marketValue.subtract(costBasis);

                    BigDecimal currentFxRate = BigDecimal.ONE;

                    if (!asset.getCurrency().equalsIgnoreCase(portfolioCurrency)) {
                        try {
                            currentFxRate = frankfurterRateService.getLatestRate(
                                    asset.getCurrency(),
                                    portfolioCurrency
                            ).rate();
                        } catch (Exception exception) {
                            throw new ResponseStatusException(
                                    HttpStatus.SERVICE_UNAVAILABLE,
                                    "FX reference rate unavailable for portfolio valuation",
                                    exception
                            );
                        }
                    }

                    if (currentFxRate == null || currentFxRate.signum() <= 0) {
                        throw new ResponseStatusException(
                                HttpStatus.SERVICE_UNAVAILABLE,
                                "Invalid FX reference rate"
                        );
                    }

                    BigDecimal convertedMarketValue =
                            marketValue.multiply(currentFxRate)
                                    .setScale(4, RoundingMode.HALF_UP);

                    BigDecimal historicalCost = holding
                            .getHistoricalCostPortfolio();

                    BigDecimal fxInclusiveProfitLoss =
                            convertedMarketValue.subtract(historicalCost);

                    return new HoldingResponse(
                            asset.getId(),
                            asset.getSymbol(),
                            asset.getName(),
                            quantity,
                            averagePrice,
                            currentPrice,
                            marketValue,
                            profitLoss,
                            asset.getCurrency(),
                            currentFxRate,
                            convertedMarketValue,
                            portfolioCurrency,
                            historicalCost,
                            fxInclusiveProfitLoss
                    );
                })
                .toList();
    }

    @Transactional(readOnly = true)
    public List<TransactionResponse> getTransactions(Long portfolioId) {
        verifyPortfolioExists(portfolioId);

        return transactionRepository
                .findByPortfolioIdOrderByCreatedAtDesc(portfolioId)
                .stream()
                .map(transaction -> new TransactionResponse(
                        transaction.getId(),
                        transaction.getAsset().getSymbol(),
                        transaction.getTransactionType(),
                        transaction.getQuantity(),
                        transaction.getPricePerUnit(),
                        transaction.getTotalAmount(),
                        transaction.getCurrency(),
                        transaction.getCreatedAt()
                ))
                .toList();
    }

    private void verifyPortfolioExists(Long portfolioId) {
        if (!portfolioRepository.existsById(portfolioId)) {
            throw new ResponseStatusException(
                    HttpStatus.NOT_FOUND,
                    "Portfolio not found"
            );
        }
    }
}

package com.portfoliolab.backend.portfolio;

import com.portfoliolab.backend.asset.Asset;
import jakarta.persistence.*;

import java.math.BigDecimal;

@Entity
@Table(
    name = "holdings",
    uniqueConstraints = @UniqueConstraint(
        columnNames = {"portfolio_id", "asset_id"}
    )
)
public class Holding {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @JoinColumn(name = "portfolio_id", nullable = false)
    private Portfolio portfolio;

    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @JoinColumn(name = "asset_id", nullable = false)
    private Asset asset;

    @Column(nullable = false, precision = 19, scale = 6)
    private BigDecimal quantity;

    @Column(
        name = "average_buy_price",
        nullable = false,
        precision = 19,
        scale = 4
    )
    private BigDecimal averageBuyPrice;

    @Column(
        name = "historical_cost_portfolio",
        nullable = false,
        precision = 19,
        scale = 4
    )
    private BigDecimal historicalCostPortfolio;

    protected Holding() {
    }

    public Holding(
            Portfolio portfolio,
            Asset asset,
            BigDecimal quantity,
            BigDecimal averageBuyPrice
    ) {
        this.portfolio = portfolio;
        this.asset = asset;
        this.quantity = quantity;
        this.averageBuyPrice = averageBuyPrice;
        this.historicalCostPortfolio = quantity
                .multiply(averageBuyPrice)
                .setScale(4, java.math.RoundingMode.HALF_UP);
    }

    public BigDecimal getHistoricalCostPortfolio() {
        return historicalCostPortfolio;
    }

    public void setHistoricalCostPortfolio(BigDecimal value) {
        if (value == null || value.signum() < 0) {
            throw new IllegalArgumentException(
                "Historical cost must be nonnegative"
            );
        }
        this.historicalCostPortfolio = value.setScale(
                4, java.math.RoundingMode.HALF_UP
        );
    }

    public Long getId() {
        return id;
    }

    public Portfolio getPortfolio() {
        return portfolio;
    }

    public Asset getAsset() {
        return asset;
    }

    public BigDecimal getQuantity() {
        return quantity;
    }

    public BigDecimal getAverageBuyPrice() {
        return averageBuyPrice;
    }

    public void updatePosition(
            BigDecimal quantity,
            BigDecimal averageBuyPrice
    ) {
        if (quantity == null || quantity.signum() <= 0) {
            throw new IllegalArgumentException(
                "Holding quantity must be positive"
            );
        }

        if (averageBuyPrice == null ||
                averageBuyPrice.signum() < 0) {
            throw new IllegalArgumentException(
                "Average buy price cannot be negative"
            );
        }

        this.quantity = quantity;
        this.averageBuyPrice = averageBuyPrice;
    }
}

package com.portfoliolab.backend.portfolio;

import com.portfoliolab.backend.asset.Asset;
import com.portfoliolab.backend.fx.FrankfurterRateService;
import com.portfoliolab.backend.market.MarketPrice;
import com.portfoliolab.backend.market.MarketService;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.web.server.ResponseStatusException;

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.time.Instant;

@Service
public class OrderPricingService {

    private final MarketService marketService;
    private final FrankfurterRateService frankfurterRateService;

    public OrderPricingService(
            MarketService marketService,
            FrankfurterRateService frankfurterRateService
    ) {
        this.marketService = marketService;
        this.frankfurterRateService = frankfurterRateService;
    }

    public Pricing calculate(
            Asset asset,
            Portfolio portfolio,
            BigDecimal quantity
    ) {
        MarketPrice quote;

        try {
            quote = marketService.getCurrentPrices()
                    .stream()
                    .filter(item -> asset.getSymbol().equals(item.symbol()))
                    .findFirst()
                    .orElse(null);
        } catch (Exception exception) {
            throw new ResponseStatusException(
                    HttpStatus.SERVICE_UNAVAILABLE,
                    "Simulated market service unavailable",
                    exception
            );
        }

        if (quote == null ||
                quote.price() == null ||
                quote.price().signum() <= 0 ||
                !asset.getCurrency().equals(quote.currency()) ||
                quote.updatedAt() == null ||
                quote.updatedAt().isBefore(Instant.now().minusSeconds(60)) ||
                quote.updatedAt().isAfter(Instant.now().plusSeconds(10))) {
            throw new ResponseStatusException(
                    HttpStatus.SERVICE_UNAVAILABLE,
                    "A valid, recent simulated market quote is unavailable"
            );
        }

        BigDecimal price = quote.price()
                .setScale(4, RoundingMode.HALF_UP);

        BigDecimal assetTotal = price.multiply(quantity)
                .setScale(4, RoundingMode.HALF_UP);

        if (assetTotal.precision() > 19 || assetTotal.scale() > 4) {
            throw new ResponseStatusException(
                    HttpStatus.BAD_REQUEST,
                    "Order total exceeds supported precision"
            );
        }

        BigDecimal fxRate;

        if (asset.getCurrency().equals(portfolio.getCurrency())) {
            fxRate = BigDecimal.ONE.setScale(8);
        } else {
            try {
                fxRate = frankfurterRateService.getLatestRate(
                        asset.getCurrency(),
                        portfolio.getCurrency()
                ).rate().setScale(8, RoundingMode.HALF_UP);
            } catch (Exception exception) {
                throw new ResponseStatusException(
                        HttpStatus.SERVICE_UNAVAILABLE,
                        "FX reference rate unavailable",
                        exception
                );
            }
        }

        if (fxRate.signum() <= 0) {
            throw new ResponseStatusException(
                    HttpStatus.SERVICE_UNAVAILABLE,
                    "Invalid FX reference rate"
            );
        }

        BigDecimal portfolioTotal = assetTotal.multiply(fxRate)
                .setScale(4, RoundingMode.HALF_UP);

        if (portfolioTotal.precision() > 19) {
            throw new ResponseStatusException(
                    HttpStatus.BAD_REQUEST,
                    "Converted order total exceeds supported precision"
            );
        }

        return new Pricing(price, assetTotal, fxRate, portfolioTotal);
    }

    public record Pricing(
            BigDecimal price,
            BigDecimal assetTotal,
            BigDecimal fxRate,
            BigDecimal portfolioTotal
    ) {}
}

package com.portfoliolab.backend.portfolio;

import com.portfoliolab.backend.asset.Asset;
import jakarta.persistence.*;

import java.math.BigDecimal;
import java.time.Instant;

@Entity
@Table(name = "transactions")
public class Transaction {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @JoinColumn(name = "portfolio_id", nullable = false)
    private Portfolio portfolio;

    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @JoinColumn(name = "asset_id", nullable = false)
    private Asset asset;

    @Column(name = "transaction_type", nullable = false, length = 4)
    private String transactionType;

    @Column(nullable = false, precision = 19, scale = 6)
    private BigDecimal quantity;

    @Column(name = "price_per_unit", nullable = false, precision = 19, scale = 4)
    private BigDecimal pricePerUnit;

    @Column(name = "total_amount", nullable = false, precision = 19, scale = 4)
    private BigDecimal totalAmount;

    @Column(name = "execution_fx_rate", nullable = false, precision = 19, scale = 8)
private BigDecimal executionFxRate;

@Column(name = "portfolio_amount", nullable = false, precision = 19, scale = 4)
private BigDecimal portfolioAmount;

    @Column(nullable = false, length = 3)
    private String currency;

    @Column(name = "created_at", nullable = false, insertable = false, updatable = false)
    private Instant createdAt;

    protected Transaction() {
    }

   public Transaction(
        Portfolio portfolio,
        Asset asset,
        String transactionType,
        BigDecimal quantity,
        BigDecimal pricePerUnit,
        BigDecimal totalAmount,
        String currency,
        BigDecimal executionFxRate,
        BigDecimal portfolioAmount
) {

package com.portfoliolab.backend.risk;

import com.portfoliolab.backend.notification.PortfolioEventOutbox;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.Map;

@Service
public class PortfolioRiskAgent {
    private final JdbcTemplate jdbc;
    private final PortfolioEventOutbox outbox;
    public PortfolioRiskAgent(JdbcTemplate jdbc, PortfolioEventOutbox outbox) {
        this.jdbc = jdbc;
        this.outbox = outbox;
    }
    @Transactional
    public Map<String, Object> analyze(long portfolioId) {
        Integer exists = jdbc.queryForObject("SELECT count(*) FROM portfolios WHERE id = ?", Integer.class, portfolioId);
        if (exists == null || exists == 0) throw new IllegalArgumentException("Portfolio not found");
        // Historical cost is already denominated in portfolio currency, unlike raw market prices.
        var positions = jdbc.query("""
                SELECT a.symbol, h.historical_cost_portfolio AS value
                FROM holdings h JOIN assets a ON a.id = h.asset_id
                WHERE h.portfolio_id = ? AND h.historical_cost_portfolio > 0
                ORDER BY h.historical_cost_portfolio DESC, a.symbol
                """, (rs, row) -> Map.<String,Object>of(
                    "symbol", rs.getString("symbol"), "value", rs.getBigDecimal("value")), portfolioId);
        if (positions.isEmpty()) return Map.of("status", "NO_HOLDINGS", "notificationCreated", false);
        BigDecimal total = positions.stream().map(p -> (BigDecimal)p.get("value"))
                .reduce(BigDecimal.ZERO, BigDecimal::add);
        if (total.signum() <= 0) return Map.of("status", "NO_COST_BASIS", "notificationCreated", false);
        var top = positions.get(0);
        String symbol = (String)top.get("symbol");
        BigDecimal percentage = ((BigDecimal)top.get("value"))
                .multiply(BigDecimal.valueOf(100)).divide(total, 1, RoundingMode.HALF_UP);
        if (percentage.compareTo(BigDecimal.valueOf(50)) < 0)
            return Map.of("status", "BELOW_THRESHOLD", "largestPosition", symbol,
                    "concentrationPercent", percentage, "notificationCreated", false);
        String band = percentage.compareTo(BigDecimal.valueOf(75)) >= 0 ? "HIGH" : "ELEVATED";
        String key = "COST_CONCENTRATION:" + symbol + ":" + band;
        // Once per day per symbol and severity; atomic suppression for concurrent requests.
        int emitted = jdbc.update("""
                INSERT INTO portfolio_risk_insights (portfolio_id, insight_key, last_emitted_at)
                VALUES (?, ?, NOW())
                ON CONFLICT (portfolio_id, insight_key) DO UPDATE
                  SET last_emitted_at = NOW()
                  WHERE portfolio_risk_insights.last_emitted_at < NOW() - INTERVAL '1 day'
                """, portfolioId, key);
        if (emitted > 0) {
            outbox.record(portfolioId, "AGENT_INSIGHT", Map.of(
                    "title", "Portfolio concentration insight",
                    "message", symbol + " represents " + percentage + "% of your invested historical cost. Consider reviewing diversification. This is a simulation, not investment advice.",
                    "targetUrl", "/portfolio"));
        }
        return Map.of("status", band, "largestPosition", symbol,
                "concentrationPercent", percentage, "notificationCreated", emitted > 0);
    }
}

package com.portfoliolab.backend.risk;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Component;
import org.springframework.transaction.PlatformTransactionManager;
import org.springframework.transaction.support.TransactionTemplate;
import org.springframework.transaction.TransactionDefinition;
import org.springframework.transaction.event.TransactionPhase;
import org.springframework.transaction.event.TransactionalEventListener;

@Component
public class PortfolioTradeRiskListener {
    private static final Logger log = LoggerFactory.getLogger(PortfolioTradeRiskListener.class);
    private final PortfolioRiskAgent agent;
    private final TransactionTemplate newTransaction;

    public PortfolioTradeRiskListener(PortfolioRiskAgent agent, PlatformTransactionManager transactionManager) {
        this.agent = agent;
        this.newTransaction = new TransactionTemplate(transactionManager);
        this.newTransaction.setPropagationBehavior(TransactionDefinition.PROPAGATION_REQUIRES_NEW);
    }

    @TransactionalEventListener(phase = TransactionPhase.AFTER_COMMIT)
    public void onTradeCommitted(PortfolioTradeCommitted event) {
        try {
            var result = newTransaction.execute(status -> agent.analyze(event.portfolioId()));
            log.info("Automatic risk analysis for portfolio {}: {}", event.portfolioId(), result);
        } catch (Exception exception) {
            // A risk analysis failure must not reverse an already committed trade.
            log.error("Automatic risk analysis failed for portfolio {}", event.portfolioId(), exception);
        }
    }
}

  _request: Request,

) {
  const { id } = await context.params;

  if (!/^\d+$/.test(id)) {
    return NextResponse.json(
      { error: "Invalid notification ID" },
      { status: 400 }
    );
  }

  try {
    const response = await fetch(
      `${backend}/api/portfolios/1/notifications/${id}/read`,
      { method: "PATCH", cache: "no-store" }
    );

    if (!response.ok) {
      return NextResponse.json(

        { status: response.status }
      );
    }

    if (response.status === 204) {
      return new Response(null, { status: 204 });
    }

    const body = await response.text();

    return new Response(body || null, {
      status: response.status,
      headers: body
        ? { "Content-Type": response.headers.get("content-type") ?? "application/json" }
        : undefined,
    });
  } catch {
    return NextResponse.json(
      { error: "Notification service unavailable" },
      { status: 503 }
    );
  }
}

import { NextRequest, NextResponse } from "next/server";

const backend = process.env.BACKEND_URL ?? "http://localhost:8080";

export async function POST(request: NextRequest) {
  let body: unknown;

  try {
    body = await request.json();
  } catch {
    return NextResponse.json(
      { error: "Invalid JSON request" },
      { status: 400 }
    );
  }

  if (
    typeof body !== "object" ||
    body === null ||
    !("assetId" in body) ||
    !("type" in body) ||
    !("quantity" in body)
  ) {
    return NextResponse.json(
      { error: "Missing order fields" },
      { status: 400 }
    );
  }

  const { assetId, type, quantity } = body;

  if (
    typeof assetId !== "number" ||
    !Number.isSafeInteger(assetId) ||
    assetId <= 0 ||
    (type !== "BUY" && type !== "SELL") ||
    typeof quantity !== "number" ||
    !Number.isSafeInteger(quantity) ||
    quantity <= 0
  ) {
    return NextResponse.json(
      { error: "Invalid asset, order type, or quantity" },
      { status: 400 }
    );
  }

  try {
    const response = await fetch(
      `${backend}/api/portfolios/1/orders/preview`,
      {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ assetId, type, quantity }),
        cache: "no-store",
        signal: AbortSignal.timeout(15000),
      }
    );

    const result = await response.text();

    return new NextResponse(result, {
      status: response.status,
      headers: {
        "Content-Type":
          response.headers.get("content-type") ??
          "application/json",
        "Cache-Control": "no-store",
      },
    });
  } catch {
    return NextResponse.json(
      {
        error: "Order preview unavailable. No trade was executed.",
      },
      { status: 503 }
    );
  }
}

"use client";

import { useEffect, useMemo, useState } from "react";
import {
  Area,
  AreaChart,
  CartesianGrid,
  ResponsiveContainer,
  Tooltip,
  XAxis,
  YAxis,
} from "recharts";

type Snapshot = {
  capturedAt: string;
  totalValue: number;
  cashBalance: number;
  holdingsValue: number;
  currency: string;
  valuationSource: string;
};

type Range = "1H" | "6H" | "1D" | "1W" | "ALL";

const ranges: { label: Range; milliseconds: number | null }[] = [
  { label: "1H", milliseconds: 60 * 60 * 1000 },
  { label: "6H", milliseconds: 6 * 60 * 60 * 1000 },
  { label: "1D", milliseconds: 24 * 60 * 60 * 1000 },
  { label: "1W", milliseconds: 7 * 24 * 60 * 60 * 1000 },
  { label: "ALL", milliseconds: null },
];

const money = (value: number) =>
  new Intl.NumberFormat("en-IN", {
    style: "currency",
    currency: "INR",
    minimumFractionDigits: 2,
    maximumFractionDigits: 2,
  }).format(value);

export default function PortfolioPerformanceChart({
  value: _value,
  ready: _ready,
}: {
  value: number;
  ready: boolean;
}) {
  const [history, setHistory] = useState<Snapshot[]>([]);
  const [range, setRange] = useState<Range>("1H");
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const [now, setNow] = useState(Date.now());

  useEffect(() => {
    let active = true;

    async function load() {
      try {
        const response = await fetch("/api/portfolio/history", {
          cache: "no-store",
        });

        if (!response.ok) {
          throw new Error("History API unavailable");
        }

        const result: Snapshot[] = await response.json();

        if (active) {
          setHistory(result);
          setNow(Date.now());
          setError(null);
        }
      } catch (cause) {
        if (active) {
          setError(
            cause instanceof Error ? cause.message : "Unable to load history"
          );
        }
      } finally {
        if (active) setLoading(false);
      }
    }

    void load();

    const interval = setInterval(() => {
      void load();
    }, 30000);

    return () => {
      active = false;
      clearInterval(interval);
    };
  }, []);

  const selectedRange = ranges.find((item) => item.label === range)!;

  const chartData = useMemo(() => {
    const cutoff =
      selectedRange.milliseconds === null
        ? null
        : now - selectedRange.milliseconds;

    return history
      .map((item) => ({
        time: new Date(item.capturedAt).getTime(),
        value: Number(item.totalValue),
      }))
      .filter(
        (item) =>
          Number.isFinite(item.time) &&
          Number.isFinite(item.value) &&
          (cutoff === null || item.time >= cutoff)
      )
      .sort((a, b) => a.time - b.time);
  }, [history, selectedRange.milliseconds, now]);

  const first = chartData[0]?.value;
  const last = chartData[chartData.length - 1]?.value;

  const change =
    first !== undefined && last !== undefined
      ? last - first
      : null;

  const changePercent =
    change !== null && first !== undefined && first !== 0
      ? (change / first) * 100
      : null;

  const positive = (change ?? 0) >= 0;

  return (
    <section className="rounded-xl border border-slate-800 bg-slate-900 p-6">
      <div className="flex flex-wrap items-start justify-between gap-4">
        <div>
          <p className="text-sm text-slate-400">
            Portfolio Value History
          </p>

          <h3 className="mt-2 text-3xl font-semibold text-white">
            {last !== undefined ? money(last) : "—"}
          </h3>

          {change !== null && (
            <p
              className={`mt-2 text-sm font-medium ${
                positive ? "text-emerald-400" : "text-red-400"
              }`}
            >
              {positive ? "+" : ""}
              {money(change)}
              {changePercent !== null &&
                ` (${positive ? "+" : ""}${changePercent.toFixed(2)}%)`}
              <span className="ml-2 font-normal text-slate-400">
                recorded value change
              </span>
            </p>
          )}
        </div>

        <div className="flex flex-wrap gap-1 rounded-lg border border-slate-700 bg-slate-950 p-1">
          {ranges.map((item) => (
            <button
              key={item.label}
              type="button"
              onClick={() => setRange(item.label)}
              aria-pressed={range === item.label}
              className={`rounded-md px-3 py-2 text-xs font-medium transition-colors ${
                range === item.label
                  ? "bg-emerald-500 text-slate-950"
                  : "text-slate-400 hover:bg-slate-800 hover:text-white"
              }`}
            >
              {item.label}
            </button>
          ))}
        </div>
      </div>

      <div className="mt-7 h-72">
        {error ? (
          <div className="flex h-full items-center justify-center text-sm text-red-400">
            {error}
          </div>
        ) : loading ? (
          <div className="flex h-full items-center justify-center text-sm text-slate-400">
            Loading recorded history...
          </div>
        ) : chartData.length < 2 ? (
          <div className="flex h-full items-center justify-center text-center text-sm text-slate-400">
            Not enough recorded observations for {range}.
            <br />
            Keep the simulator and backend running.
          </div>
        ) : (
          <ResponsiveContainer width="100%" height="100%">
            <AreaChart data={chartData}>
              <defs>
                <linearGradient
                  id="portfolioHistoryFill"
                  x1="0"
                  y1="0"
                  x2="0"
                  y2="1"

export type ValuationHolding = {
  quantity: number;
  averageBuyPrice: number;
  currentPrice: number;
  currentFxRate: number;
  historicalCostInPortfolioCurrency: number;
};

export function valueHolding<T extends ValuationHolding>(
  holding: T,
  currentPrice: number
) {
  const fxRate = holding.currentFxRate;
  const historicalCost =
    holding.historicalCostInPortfolioCurrency;

  if (!Number.isFinite(fxRate) || fxRate <= 0) {
    throw new Error("Invalid FX rate for portfolio valuation");
  }

  if (!Number.isFinite(currentPrice) || currentPrice < 0) {
    throw new Error("Invalid market price for portfolio valuation");
  }

  if (
    !Number.isFinite(historicalCost) ||
    historicalCost < 0
  ) {
    throw new Error("Invalid historical cost for portfolio valuation");
  }

  const marketValue = holding.quantity * currentPrice;

  const assetCurrencyCostBasis =
    holding.quantity * holding.averageBuyPrice;

  const marketValueInPortfolioCurrency =
    marketValue * fxRate;

  const unrealizedProfitLossInPortfolioCurrency =
    marketValueInPortfolioCurrency - historicalCost;

  return {
    ...holding,
    currentPrice,
    marketValue,

    // Price-only P&L in the asset's own currency.
    unrealizedProfitLoss:
      marketValue - assetCurrencyCostBasis,

    marketValueInPortfolioCurrency,

    // Historical cost remains fixed at executed trade FX.
    historicalCostInPortfolioCurrency: historicalCost,

    // Includes both market-price and FX movements.
    unrealizedProfitLossInPortfolioCurrency,

    // Retained for price-only attribution at current FX.
    costBasisAtCurrentFx:
      assetCurrencyCostBasis * fxRate,

    unrealizedProfitLossAtCurrentFx:
      (marketValue - assetCurrencyCostBasis) * fxRate,
  };
}

export function summarizeHoldings<
  T extends {
    marketValueInPortfolioCurrency: number;
    historicalCostInPortfolioCurrency: number;
  }
>(
  holdings: T[],
  cashBalance: number
) {
  const holdingsValue = holdings.reduce(
    (sum, holding) =>
      sum + holding.marketValueInPortfolioCurrency,
    0
  );

  const investedAmount = holdings.reduce(
    (sum, holding) =>
      sum + holding.historicalCostInPortfolioCurrency,
    0
  );

  return {
    holdingsValue,
    investedAmount,
    unrealizedProfitLoss:
      holdingsValue - investedAmount,
    totalValue: cashBalance + holdingsValue,
  };
}

export function formatMoney(value: number, currency: string) {
  return new Intl.NumberFormat("en-IN", {
    style: "currency",
    currency,
    minimumFractionDigits: 2,
    maximumFractionDigits: 2,
  }).format(value);

export function usePortfolioData() {
  const [data, setData] = useState<PortfolioData | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const { prices, status } = useMarketPrices();

  const refresh = useCallback(async () => {
    const response = await fetch("/api/portfolio", {
      cache: "no-store",
    });

    if (!response.ok) {
      throw new Error("Unable to load portfolio data");
    }

    const result: PortfolioData = await response.json();
    setData(result);
    setError(null);
  }, []);

  useEffect(() => {
    let active = true;

    async function load() {
      try {
        const response = await fetch("/api/portfolio", {
          cache: "no-store",
        });

        if (!response.ok) {
          throw new Error("Unable to load portfolio data");
        }

        const result: PortfolioData = await response.json();

        if (active) {
          setData(result);
          setError(null);
        }
      } catch (err) {
        if (active) {
          setError(
            err instanceof Error ? err.message : "Unknown error"
          );
        }
      } finally {
        if (active) setLoading(false);
      }
    }

    void load();

    return () => {
      active = false;
    };
  }, []);

  const livePriceBySymbol = new Map(
    prices.map((price) => [price.symbol, price])
  );

  const holdings = (data?.holdings ?? []).map((holding) => {
    const quote = livePriceBySymbol.get(holding.symbol);

    const priceIsLive =
      status === "connected" &&
      quote?.currency === holding.currency;

    const currentPrice = priceIsLive
      ? quote.price
      : holding.currentPrice;

    const fxRate =
      holding.currency === data?.portfolio.currency
        ? 1
        : holding.currentFxRate;

    return {
      ...valueHolding(
        { ...holding, currentFxRate: fxRate },
        currentPrice
      ),
      priceIsLive,
    };
  });

  const {
    holdingsValue,
    investedAmount,
    unrealizedProfitLoss,
    totalValue,
  } = summarizeHoldings(
    holdings,
    data?.portfolio.cashBalance ?? 0
  );
  const isPortfolioLive =
    holdings.length > 0 &&
    holdings.every((holding) => holding.priceIsLive);

  return {
    portfolio: data?.portfolio ?? null,
    holdings,
    transactions: data?.transactions ?? [],
    loading,
    error,
    refresh,
    holdingsValue,
    investedAmount,
    unrealizedProfitLoss,
    totalValue,
    isPortfolioLive,
  };
}
