"use client";

import { useState } from "react";
import {
  formatMoney,
  type Holding,
  type Portfolio,

type Props = {
  portfolio: Portfolio;
  holdings: Holding[];
  totalValue: number;
  isPortfolioLive: boolean;

const colors = [
  "#10b981",
  "#38bdf8",
  "#a78bfa",
  "#f59e0b",
  "#f472b6",
  "#22d3ee",
  "#fb7185",
];

export default function PortfolioAllocation({
  portfolio,
  holdings,
  totalValue,
  isPortfolioLive,
}: Props) {
  const [selectedKey, setSelectedKey] = useState<string>("cash");

  const positions = holdings
    .filter(
      (holding) =>
        Number.isFinite(holding.marketValueInPortfolioCurrency) &&
        holding.marketValueInPortfolioCurrency > 0
    )
    .map((holding, index) => ({
      key: String(holding.assetId),
      label: holding.symbol,
      name: holding.name,
      value: holding.marketValueInPortfolioCurrency,
      color: colors[index % colors.length],
    }));

  const items = [
    ...positions,
    {
      key: "cash",
      label: "Cash",
      name: "Available simulated cash",
      value: Math.max(0, portfolio.cashBalance),
      color: "#64748b",
    },
  ];

  const allocationTotal = items.reduce(
    (sum, item) => sum + item.value,
    0
  );

  const selected =
    items.find((item) => item.key === selectedKey) ?? items[0];

  if (!Number.isFinite(allocationTotal) || allocationTotal <= 0) {
    return (
      <section className="rounded-xl border border-slate-800 bg-slate-900 p-6">
        <h3 className="text-lg font-semibold">Portfolio Allocation</h3>
        <p className="mt-3 text-sm text-slate-400">
          Allocation will appear when your portfolio has a positive value.
        </p>
      </section>
    );
  }

  return (
    <section className="rounded-xl border border-slate-800 bg-slate-900 p-6">
      <div className="flex flex-wrap items-start justify-between gap-4">
        <div>
          <h3 className="text-lg font-semibold">
            Portfolio Allocation
          </h3>
          <p className="mt-1 text-sm text-slate-400">
            Select a segment to explore your portfolio composition.
          </p>
        </div>

        <span className="rounded-full border border-slate-700 px-3 py-1 text-xs text-slate-400">
          {isPortfolioLive ? "Live simulated prices" : "Mixed/reference prices"}
        </span>
      </div>

      <p className="mt-5 text-xs text-slate-400">
        Total portfolio value
      </p>
      <p className="mt-1 text-2xl font-semibold tabular-nums">
        {formatMoney(totalValue, portfolio.currency)}
      </p>

      <div
        className="mt-5 flex h-5 overflow-hidden rounded-full bg-slate-800"
        aria-label="Portfolio allocation"
      >
        {items
          .filter((item) => item.value > 0)
          .map((item) => (
            <button
              key={item.key}
              type="button"
              title={`${item.label}: ${(
                (item.value / allocationTotal) *
                100
              ).toFixed(1)}%`}
              aria-label={`Select ${item.label} allocation`}
              aria-pressed={selected.key === item.key}
              onClick={() => setSelectedKey(item.key)}
              className={`h-full cursor-pointer transition-opacity hover:opacity-75 focus-visible:outline-2 focus-visible:outline-white ${

"use client";

import { useEffect, useMemo, useState } from "react";
import {
  Area,
  AreaChart,
  CartesianGrid,
  ResponsiveContainer,
  Tooltip,
  XAxis,
  YAxis,
} from "recharts";

type Snapshot = {
  capturedAt: string;
  totalValue: number;
  cashBalance: number;
  holdingsValue: number;
  currency: string;
  valuationSource: string;
};

type Range = "1H" | "6H" | "1D" | "1W" | "ALL";

const ranges: { label: Range; milliseconds: number | null }[] = [
  { label: "1H", milliseconds: 60 * 60 * 1000 },
  { label: "6H", milliseconds: 6 * 60 * 60 * 1000 },
  { label: "1D", milliseconds: 24 * 60 * 60 * 1000 },
  { label: "1W", milliseconds: 7 * 24 * 60 * 60 * 1000 },
  { label: "ALL", milliseconds: null },
];

const money = (value: number) =>
  new Intl.NumberFormat("en-IN", {
    style: "currency",
    currency: "INR",
    minimumFractionDigits: 2,
    maximumFractionDigits: 2,
  }).format(value);

export default function PortfolioPerformanceChart({
  value: _value,
  ready: _ready,
}: {
  value: number;
  ready: boolean;
}) {
  const [history, setHistory] = useState<Snapshot[]>([]);
  const [range, setRange] = useState<Range>("1H");
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const [now, setNow] = useState(Date.now());

  useEffect(() => {
    let active = true;

    async function load() {
      try {
        const response = await fetch("/api/portfolio/history", {
          cache: "no-store",
        });

        if (!response.ok) {
          throw new Error("History API unavailable");
        }

        const result: Snapshot[] = await response.json();

        if (active) {
          setHistory(result);
          setNow(Date.now());
          setError(null);
        }
      } catch (cause) {
        if (active) {
          setError(
            cause instanceof Error ? cause.message : "Unable to load history"
          );
        }
      } finally {
        if (active) setLoading(false);
      }
    }

    void load();

    const interval = setInterval(() => {
      void load();
    }, 30000);

    return () => {
      active = false;
      clearInterval(interval);
    };
  }, []);

  const selectedRange = ranges.find((item) => item.label === range)!;

              </p>
              <p className="text-xs text-slate-400">
                Business Rules Agent:{" "}
                {currentValidation.overallStatus}
              </p>
              <p className="text-xs text-slate-400">
                Buying power:{" "}
                {currentValidation.side === "BUY"
                  ? orderPreview.sufficientCash === true
                    ? "Sufficient"
                    : orderPreview.sufficientCash === false
                      ? "Insufficient"
                      : "Unknown"
                  : "Not applicable to SELL"}
              </p>
            </div>
          )}

          <div className="space-y-2 border-t border-slate-800 pt-3">
            <h5 className="text-sm font-semibold">
              Estimated order cost
            </h5>
            <p className="text-sm text-slate-300">
              Estimated asset cost:{" "}
              {formatMoney(
                orderPreview.estimatedAssetTotal,
                orderPreview.assetCurrency
              )}
            </p>
            <p className="text-sm text-slate-300">
              Reference FX rate:{" "}
              {orderPreview.fxRate.toFixed(6)}{" "}
              {orderPreview.portfolioCurrency} per{" "}
              {orderPreview.assetCurrency}
            </p>
            <p className="text-sm text-slate-300">
              Estimated portfolio cost:{" "}
              {formatMoney(
                orderPreview.estimatedPortfolioTotal,
                orderPreview.portfolioCurrency
              )}
            </p>
            {type === "BUY" && (
              <>
                <p className="text-sm text-slate-300">
                  Available cash:{" "}
                  {formatMoney(
                    orderPreview.availableCash,
                    orderPreview.portfolioCurrency
                  )}
                </p>
                <p
                  className={
                    orderPreview.sufficientCash
                      ? "text-sm text-emerald-300"
                      : "text-sm text-red-300"
                  }
                >
                  Buying power:{" "}
                  {orderPreview.sufficientCash
                    ? "Sufficient"
                    : "Insufficient"}
                </p>
              </>
            )}
            <p className="text-xs text-slate-500">
              {orderPreview.disclaimer}
            </p>
          </div>

          {impactCanBeShown && selectedAsset && estimatedCashAfter !== null && (
            <div className="rounded-lg border border-slate-700 bg-slate-900 p-4">
              <h5 className="font-semibold">Estimated portfolio impact</h5>
              <p className="mt-1 text-xs text-slate-400">
                Hypothetical change if this simulated order executes at the
                previewed amount.
              </p>

              <div className="mt-4 grid gap-4 sm:grid-cols-2">
                <div>
                  <p className="text-xs text-slate-400">Available cash</p>
                  <p className="mt-1 text-sm tabular-nums text-slate-300">
                    {formatMoney(portfolio.cashBalance, portfolio.currency)}
                  </p>
                  <p className="mt-1 font-semibold tabular-nums text-emerald-300">
                    → {formatMoney(estimatedCashAfter, portfolio.currency)}
                  </p>
                </div>

                <div>
                  <p className="text-xs text-slate-400">
                    {selectedAsset.symbol} allocation
                  </p>
                  <p className="mt-1 text-sm tabular-nums text-slate-300">
                    {currentAllocation.toFixed(1)}%
                  </p>
                  <p className="mt-1 font-semibold tabular-nums text-emerald-300">
                    → {estimatedAllocation.toFixed(1)}%
                  </p>
                </div>
              </div>

              <p className="mt-4 text-xs leading-5 text-slate-500">
                Illustrative allocation using current portfolio valuations and
                the backend's estimated trade amount. Execution prices, FX,
                charges, and portfolio values may differ. No trade is executed
                by this preview.
              </p>
            </div>
          )}

          <details className="rounded-lg border border-slate-800 bg-slate-900/50">
            <summary className="cursor-pointer px-4 py-3 text-sm font-medium text-slate-200 hover:text-emerald-300">
              Detailed market-rule checks ({currentValidation.checks.length})
            </summary>

            <div className="space-y-3 border-t border-slate-800 px-4 py-3">
              {currentValidation.checks.map((check) => (
                <div
                  key={check.code}
                  className="border-b border-slate-800 pb-3 last:border-b-0 last:pb-0"
                >
                  <div className="flex items-center justify-between gap-2">
                    <span className="text-sm font-medium">
                      {check.code.replaceAll("_", " ")}
                    </span>
                    <span
                      className={`text-xs font-semibold ${
                        check.status === "FAIL"
                          ? "text-red-300"
                          : check.status === "PASS"
                            ? "text-emerald-300"
                            : "text-amber-300"
                      }`}
                    >
                      {check.status.replaceAll("_", " ")}
                    </span>
                  </div>
                  <p className="mt-1 text-xs text-slate-400">
                    {check.explanation}
                  </p>
                </div>
              ))}
            </div>
          </details>

          <p className="text-xs leading-5 text-slate-500">
            {currentValidation.disclaimer}
          </p>
          <p className="text-xs text-amber-300">
            Validation is informational and does not reserve cash,
            lock a price, or execute an order.
          </p>
        </div>
      )}
        </div>
      </div>

      {message && (
        <p role="status" className="mt-4 text-sm text-emerald-400">
          {message}
        </p>
      )}

      {error && (
        <p role="alert" className="mt-4 text-sm text-red-400">
          {error}
        </p>
      )}
    </section>
  );
}

"use client";

import { useCallback, useEffect, useState } from "react";
import { useMarketPrices } from "@/components/MarketPriceProvider";
import {
  valueHolding,
  summarizeHoldings,
} from "@/components/portfolioValuation";

export type Portfolio = {
  id: number;
  name: string;
  cashBalance: number;
  currency: string;
};

export type Holding = {
  assetId: number;
  symbol: string;
  name: string;
  quantity: number;
  averageBuyPrice: number;
  currentPrice: number;
  marketValue: number;
  unrealizedProfitLoss: number;
  currency: string;
  currentFxRate: number;
  marketValueInPortfolioCurrency: number;
  portfolioCurrency: string;
  historicalCostInPortfolioCurrency: number;
  unrealizedProfitLossInPortfolioCurrency: number;
};

export type Transaction = {
  id: number;
  assetSymbol: string;
  type: string;
  quantity: number;
  pricePerUnit: number;
  totalAmount: number;
  currency: string;
  createdAt: string;
};

type PortfolioData = {
  portfolio: Portfolio;
  holdings: Holding[];
  transactions: Transaction[];
};

export function formatMoney(value: number, currency: string) {
  return new Intl.NumberFormat("en-IN", {
    style: "currency",
    currency,
    minimumFractionDigits: 2,
    maximumFractionDigits: 2,
  }).format(value);
}

export function usePortfolioData() {
  const [data, setData] = useState<PortfolioData | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  const { prices, status } = useMarketPrices();

  const refresh = useCallback(async () => {
    const response = await fetch("/api/portfolio", {
      cache: "no-store",
    });

    if (!response.ok) {
      throw new Error("Unable to load portfolio data");
    }

    const result: PortfolioData = await response.json();
    setData(result);
    setError(null);
  }, []);

  useEffect(() => {
    let active = true;

    async function load() {
      try {
        const response = await fetch("/api/portfolio", {
          cache: "no-store",
        });

        if (!response.ok) {
          throw new Error("Unable to load portfolio data");
        }

        const result: PortfolioData = await response.json();

        if (active) {
          setData(result);
          setError(null);
        }
      } catch (err) {
        if (active) {
          setError(
            err instanceof Error ? err.message : "Unknown error"
          );
        }
      } finally {
        if (active) setLoading(false);
      }
    }

    void load();

    return () => {
      active = false;
    };
  }, []);

  const livePriceBySymbol = new Map(
    prices.map((price) => [price.symbol, price])
  );

  const holdings = (data?.holdings ?? []).map((holding) => {
    const quote = livePriceBySymbol.get(holding.symbol);

    const priceIsLive =
      status === "connected" &&
      quote?.currency === holding.currency;

    const currentPrice = priceIsLive
      ? quote.price
      : holding.currentPrice;

    const fxRate =
      holding.currency === data?.portfolio.currency
        ? 1
        : holding.currentFxRate;

    return {
      ...valueHolding(
        { ...holding, currentFxRate: fxRate },
        currentPrice
      ),
      priceIsLive,
    };
  });

  const {
    holdingsValue,
    investedAmount,
    unrealizedProfitLoss,
    totalValue,
  } = summarizeHoldings(
    holdings,
    data?.portfolio.cashBalance ?? 0
  );
  const isPortfolioLive =
    holdings.length > 0 &&
    holdings.every((holding) => holding.priceIsLive);

  return {
    portfolio: data?.portfolio ?? null,
    holdings,
    transactions: data?.transactions ?? [],
    loading,
    error,
    refresh,
    holdingsValue,
    investedAmount,
    unrealizedProfitLoss,
    totalValue,
    isPortfolioLive,
  };
}
