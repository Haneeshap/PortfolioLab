package com.portfoliolab.backend.businessrules;

import java.math.BigDecimal;
import java.util.ArrayList;
import java.util.List;
import org.springframework.stereotype.Service;

@Service
public class MarketRulesService {

    public record RuleCheck(
            String code,
            String status,
            String explanation
    ) {}

    public record Evaluation(
            String market,
            String currency,
            String side,
            String overallStatus,
            List<RuleCheck> checks,
            String disclaimer
    ) {}

    public Evaluation evaluate(
            String market,
            String side,
            long quantity,
            BigDecimal price,
            BigDecimal availableCash,
            Long availableShares,
            BigDecimal estimatedCharges,
            BigDecimal lowerPriceLimit,
            BigDecimal upperPriceLimit,
            Long quantityIncrement) {

        if (!List.of("NSE", "BSE", "NYSE", "NASDAQ").contains(market)) {
            throw new IllegalArgumentException("Unsupported market");
        }

        if (!List.of("BUY", "SELL").contains(side)) {
            throw new IllegalArgumentException("Unsupported transaction side");
        }

        if (quantity <= 0 || price == null ||
                price.signum() <= 0) {
            throw new IllegalArgumentException(
                    "Quantity and price must be positive");
        }

        if (availableCash != null && availableCash.signum() < 0 ||
                availableShares != null && availableShares < 0 ||
                estimatedCharges != null && estimatedCharges.signum() < 0 ||
                quantityIncrement != null && quantityIncrement <= 0) {
            throw new IllegalArgumentException("Invalid simulation input");
        }

        if (lowerPriceLimit != null && lowerPriceLimit.signum() <= 0 ||
                upperPriceLimit != null && upperPriceLimit.signum() <= 0 ||
                lowerPriceLimit != null && upperPriceLimit != null &&
                lowerPriceLimit.compareTo(upperPriceLimit) > 0) {
            throw new IllegalArgumentException("Invalid price limits");
        }

        String currency = List.of("NSE", "BSE").contains(market)
                ? "INR" : "USD";

        List<RuleCheck> checks = new ArrayList<>();

        if ("BUY".equals(side)) {
            if (availableCash == null || estimatedCharges == null) {
                checks.add(new RuleCheck(
                        "BUYING_POWER", "NEEDS_DATA",
                        "Available cash and estimated charges are required."
                ));
            } else {
                BigDecimal required = price
                        .multiply(BigDecimal.valueOf(quantity))
                        .add(estimatedCharges);

                boolean passes = availableCash.compareTo(required) >= 0;

                checks.add(new RuleCheck(
                        "BUYING_POWER",
                        passes ? "PASS" : "FAIL",
                        "Required " + currency + " " + required
                                + "; available " + currency + " "
                                + availableCash + "."
                ));
            }
        } else {
            if (availableShares == null) {
                checks.add(new RuleCheck(
                        "SELL_AVAILABILITY", "NEEDS_DATA",
                        "Available delivery shares are required."
                ));
            } else {
                checks.add(new RuleCheck(
                        "SELL_AVAILABILITY",
                        availableShares >= quantity ? "PASS" : "FAIL",
                        "Requested " + quantity
                                + " shares; available "
                                + availableShares + "."
                ));
            }
        }

        if (lowerPriceLimit == null || upperPriceLimit == null) {
            checks.add(new RuleCheck(
                    "PRICE_RESTRICTIONS", "NEEDS_DATA",
                    "Applicable security-specific price restrictions "
                            + "have not been supplied."
            ));
        } else {
            boolean passes =
                    price.compareTo(lowerPriceLimit) >= 0 &&
                    price.compareTo(upperPriceLimit) <= 0;

            checks.add(new RuleCheck(
                    "PRICE_RESTRICTIONS",
                    passes ? "PASS" : "FAIL",
                    "Order price " + price
                            + "; supplied permitted range "
                            + lowerPriceLimit + " to "
                            + upperPriceLimit + "."
            ));
        }

        long increment = quantityIncrement == null ? 1 : quantityIncrement;

        checks.add(new RuleCheck(
                "QUANTITY_INCREMENT",
                quantity % increment == 0 ? "PASS" : "FAIL",
                "Quantity must be a multiple of " + increment
                        + " for this simulated instrument."
        ));

        boolean failed = checks.stream()
                .anyMatch(c -> "FAIL".equals(c.status()));

        boolean missing = checks.stream()
                .anyMatch(c -> "NEEDS_DATA".equals(c.status()));

        String overall = failed ? "FAIL" :
                missing ? "NEEDS_DATA" : "PASS";

        return new Evaluation(
                market,
                currency,
                side,
                overall,
                List.copyOf(checks),
                "Educational simulation only. These checks do not "
                        + "verify live exchange rules, trading halts, "
                        + "broker restrictions, or actual order eligibility."
        );
    }
}

package com.portfoliolab.backend.history;

import com.portfoliolab.backend.market.MarketPrice;
import com.portfoliolab.backend.market.MarketService;

import com.portfoliolab.backend.portfolio.PortfolioRepository;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Service;

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.List;
import java.util.Map;
import java.util.function.Function;
import java.util.stream.Collectors;

@Service
public class PortfolioHistoryRecorder {
    private static final Logger log =
            LoggerFactory.getLogger(PortfolioHistoryRecorder.class);

    private final PortfolioRepository portfolios;
    private final HoldingRepository holdings;
    private final MarketService market;
    private final JdbcTemplate jdbc;

    public PortfolioHistoryRecorder(
            PortfolioRepository portfolios,
            HoldingRepository holdings,
            MarketService market,
            JdbcTemplate jdbc
    ) {
        this.portfolios = portfolios;
        this.holdings = holdings;
        this.market = market;
        this.jdbc = jdbc;
    }

    @Scheduled(fixedDelay = 30000, initialDelay = 5000)
    public void record() {
        List<Portfolio> allPortfolios = portfolios.findAll();

        if (allPortfolios.isEmpty()) {
            return;
        }

        Map<String, MarketPrice> quotes;
        try {
            quotes = market.getCurrentPrices().stream()
                    .collect(Collectors.toMap(
                            MarketPrice::symbol,
                            Function.identity(),
                            (first, second) -> second
                    ));
        } catch (Exception exception) {
            log.warn("Skipping history snapshot: market unavailable");
            return;
        }

        for (Portfolio portfolio : allPortfolios) {
            try {
                recordPortfolio(portfolio, quotes);
            } catch (Exception exception) {
                log.warn(
                        "Could not record portfolio {}: {}",
                        portfolio.getId(),
                        exception.getMessage()
                );
            }
        }
    }

    private void recordPortfolio(
            Portfolio portfolio,
            Map<String, MarketPrice> quotes
    ) {
        List<Holding> positions =
                holdings.findByPortfolioId(portfolio.getId());

        BigDecimal holdingsValue = BigDecimal.ZERO;

        for (Holding holding : positions) {
            String symbol = holding.getAsset().getSymbol();
            String currency = holding.getAsset().getCurrency();

            MarketPrice quote = quotes.get(symbol);

            if (quote == null
                    || quote.price() == null
                    || quote.price().signum() <= 0
                    || !currency.equals(quote.currency())
                    || !portfolio.getCurrency().equals(currency)) {
                log.warn(
                        "Skipping snapshot for portfolio {}: invalid or missing quote for {}",
                        portfolio.getId(),
                        symbol
                );
                return;
            }

            holdingsValue = holdingsValue.add(
                    holding.getQuantity().multiply(quote.price())
            );
        }

        BigDecimal cash = portfolio.getCashBalance();
        BigDecimal total = cash.add(holdingsValue);

        jdbc.update("""
                INSERT INTO portfolio_snapshots
                (portfolio_id, cash_balance, holdings_value,
                 total_value, currency, valuation_source)
                VALUES (?, ?, ?, ?, ?, ?)
                """,
                portfolio.getId(),
                cash,
                holdingsValue.setScale(4, RoundingMode.HALF_UP),
                total.setScale(4, RoundingMode.HALF_UP),
                portfolio.getCurrency(),
                "GO_SIMULATOR"
        );
    }
}

   tx.executeWithoutResult(status->{

     jdbc.update("INSERT INTO market_price_samples(symbol,currency,price) VALUES(?,?,?)",q.symbol(),q.currency(),current);

     check(q.symbol(),q.currency(),current,300,new BigDecimal("6.0"));
    }
   });

 private void check(String symbol,String currency,BigDecimal current,int seconds,BigDecimal threshold){
  // Nearest sample around the beginning of the window (Â±15s).
  List<BigDecimal> candidates=jdbc.query("""
   SELECT price FROM market_price_samples WHERE symbol=? AND currency=?
   AND observed_at BETWEEN NOW() - (? * INTERVAL '1 second') - INTERVAL '15 seconds'
                       AND NOW() - (? * INTERVAL '1 second') + INTERVAL '15 seconds'
   ORDER BY ABS(EXTRACT(EPOCH FROM (observed_at-(NOW()-(? * INTERVAL '1 second'))))) ASC,id ASC LIMIT 1
   """,(rs,n)->rs.getBigDecimal(1),symbol,currency,seconds,seconds,seconds);
  if(candidates.isEmpty()||candidates.get(0).signum()<=0)return;
  BigDecimal start=candidates.get(0);
  BigDecimal percent=current.subtract(start).multiply(new BigDecimal("100"))
    .divide(start,4,RoundingMode.HALF_UP);
  if(percent.abs().compareTo(threshold)<0)return;
  String type=percent.signum()>0?"SURGE":"DROP";
  // Suppress repeats across observation windows for the same asset/direction.
  Integer recent=jdbc.queryForObject("""
   SELECT count(*) FROM market_intelligence_events WHERE symbol=? AND event_type=?
   AND created_at > NOW()-INTERVAL '10 minutes'
   """,Integer.class,symbol,type);
  if(recent!=null&&recent>0)return;
  String window=seconds==60?"1 minute":"5 minutes";
  String headline=symbol+(percent.signum()>0?" rises sharply":" falls sharply")+" in simulated trading";
  String explanation=String.format(Locale.ROOT,
   "%s moved %s%% over %s, from %s to %s %s. Flagged because the absolute move exceeded %s%%. All prices are simulated.",
   symbol,percent.toPlainString(),window,start.toPlainString(),current.toPlainString(),currency,threshold.toPlainString());
  Long id=jdbc.queryForObject("""
   INSERT INTO market_intelligence_events(symbol,currency,event_type,window_seconds,start_price,end_price,change_percent,headline,explanation)
   VALUES(?,?,?,?,?,?,?,?,?) RETURNING id
   """,Long.class,symbol,currency,type,seconds,start,current,percent,headline,explanation);
  try {
   String payload=mapper.writeValueAsString(Map.of("eventType","MARKET_INTELLIGENCE","intelligenceId",id,
    "symbol",symbol,"currency",currency,"direction",type,"windowSeconds",seconds,
    "changePercent",percent,"headline",headline,"explanation",explanation));
   jdbc.update("INSERT INTO market_intelligence_outbox(intelligence_id,payload) VALUES(?,?)",id,payload);
  } catch(Exception e){throw new IllegalStateException("Unable to serialize market intelligence event",e);}
 }

package com.portfoliolab.backend.portfolio;

import com.portfoliolab.backend.asset.Asset;
import com.portfoliolab.backend.fx.FrankfurterRateService;
import com.portfoliolab.backend.market.MarketPrice;
import com.portfoliolab.backend.market.MarketService;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.web.server.ResponseStatusException;

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.time.Instant;

@Service
public class OrderPricingService {

    private final MarketService marketService;
    private final FrankfurterRateService frankfurterRateService;

    public OrderPricingService(
            MarketService marketService,
            FrankfurterRateService frankfurterRateService
    ) {
        this.marketService = marketService;
        this.frankfurterRateService = frankfurterRateService;
    }

    public Pricing calculate(
            Asset asset,
            Portfolio portfolio,
            BigDecimal quantity
    ) {
        MarketPrice quote;

        try {
            quote = marketService.getCurrentPrices()
                    .stream()
                    .filter(item -> asset.getSymbol().equals(item.symbol()))
                    .findFirst()
                    .orElse(null);
        } catch (Exception exception) {
            throw new ResponseStatusException(
                    HttpStatus.SERVICE_UNAVAILABLE,
                    "Simulated market service unavailable",
                    exception
            );
        }

        if (quote == null ||
                quote.price() == null ||
                quote.price().signum() <= 0 ||
                !asset.getCurrency().equals(quote.currency()) ||
                quote.updatedAt() == null ||
                quote.updatedAt().isBefore(Instant.now().minusSeconds(60)) ||
                quote.updatedAt().isAfter(Instant.now().plusSeconds(10))) {
            throw new ResponseStatusException(
                    HttpStatus.SERVICE_UNAVAILABLE,
                    "A valid, recent simulated market quote is unavailable"
            );
        }

        BigDecimal price = quote.price()
                .setScale(4, RoundingMode.HALF_UP);

        BigDecimal assetTotal = price.multiply(quantity);

        if (assetTotal.precision() > 19 || assetTotal.scale() > 4) {
            throw new ResponseStatusException(
                    HttpStatus.BAD_REQUEST,
                    "Order total exceeds supported precision"
            );
        }

        BigDecimal fxRate;

        if (asset.getCurrency().equals(portfolio.getCurrency())) {
            fxRate = BigDecimal.ONE.setScale(8);
        } else {
            try {
                fxRate = frankfurterRateService.getLatestRate(
                        asset.getCurrency(),
                        portfolio.getCurrency()
                ).rate().setScale(8, RoundingMode.HALF_UP);
            } catch (Exception exception) {
                throw new ResponseStatusException(
                        HttpStatus.SERVICE_UNAVAILABLE,
                        "FX reference rate unavailable",
                        exception
                );
            }
        }

        if (fxRate.signum() <= 0) {
            throw new ResponseStatusException(
                    HttpStatus.SERVICE_UNAVAILABLE,
                    "Invalid FX reference rate"
            );
        }

        BigDecimal portfolioTotal = assetTotal.multiply(fxRate)
                .setScale(4, RoundingMode.HALF_UP);

        if (portfolioTotal.precision() > 19) {
            throw new ResponseStatusException(
                    HttpStatus.BAD_REQUEST,
                    "Converted order total exceeds supported precision"
            );
        }

        return new Pricing(price, assetTotal, fxRate, portfolioTotal);
    }

    public record Pricing(
            BigDecimal price,
            BigDecimal assetTotal,
            BigDecimal fxRate,
            BigDecimal portfolioTotal
    ) {}
}

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.server.ResponseStatusException;

import java.math.BigDecimal;
import java.math.RoundingMode;
import java.util.Locale;
@Service
public class OrderService {

    private final PortfolioRepository portfolioRepository;
    private final AssetRepository assetRepository;
    private final HoldingRepository holdingRepository;
    private final TransactionRepository transactionRepository;
    private final OrderPricingService pricingService;
    private final com.portfoliolab.backend.risk.PreTradeRiskService preTradeRiskService;
    private final NotificationOutbox notificationOutbox;
    private final ApplicationEventPublisher eventPublisher;

    public OrderService(
            PortfolioRepository portfolioRepository,
            AssetRepository assetRepository,
            HoldingRepository holdingRepository,
            TransactionRepository transactionRepository,
            OrderPricingService pricingService,
            com.portfoliolab.backend.risk.PreTradeRiskService preTradeRiskService,
            NotificationOutbox notificationOutbox,
            ApplicationEventPublisher eventPublisher
    ) {
        this.portfolioRepository = portfolioRepository;
        this.assetRepository = assetRepository;
        this.holdingRepository = holdingRepository;
        this.transactionRepository = transactionRepository;
        this.pricingService = pricingService;
        this.preTradeRiskService = preTradeRiskService;
        this.notificationOutbox = notificationOutbox;
        this.eventPublisher = eventPublisher;
    }

    @Transactional
    public OrderResponse placeOrder(Long portfolioId, OrderRequest request) {

        if (request == null ||
                request.assetId() == null ||
                request.type() == null ||
                request.quantity() == null) {
            throw new ResponseStatusException(
                    HttpStatus.BAD_REQUEST,
                    "Asset ID, order type, and quantity are required"
            );
        }

        String type = request.type().trim().toUpperCase(Locale.ROOT);
        BigDecimal quantity = request.quantity();

        if (!type.equals("BUY") && !type.equals("SELL")) {
            throw new ResponseStatusException(
                    HttpStatus.BAD_REQUEST,
                    "Order type must be BUY or SELL"
            );
        }

        if (quantity.signum() <= 0 ||
                quantity.stripTrailingZeros().scale() > 0 ||
                quantity.precision() > 13) {
            throw new ResponseStatusException(
                    HttpStatus.BAD_REQUEST,
                    "Quantity must be a positive whole number of at most 13 digits"
            );
        }

        Portfolio portfolio = portfolioRepository.findByIdForUpdate(portfolioId)
                .orElseThrow(() -> new ResponseStatusException(
                        HttpStatus.NOT_FOUND,
                        "Portfolio not found"
                ));

        Asset asset = assetRepository.findById(request.assetId())
                .orElseThrow(() -> new ResponseStatusException(
                        HttpStatus.NOT_FOUND,
                        "Asset not found"
                ));



        OrderPricingService.Pricing pricing =
                pricingService.calculate(asset, portfolio, quantity);

        BigDecimal price = pricing.price();
        BigDecimal total = pricing.assetTotal();
        BigDecimal executionFxRate = pricing.fxRate();
        BigDecimal portfolioAmount = pricing.portfolioTotal();

        Holding holding = holdingRepository
                .findByPortfolioIdAndAssetId(portfolioId, asset.getId())
                .orElse(null);

        preTradeRiskService.enforce(portfolio, asset, type, quantity, portfolioAmount);

        if (type.equals("BUY")) {
            processBuy(portfolio, asset, holding, quantity, price, portfolioAmount);
        } else {
            processSell(portfolio, holding, quantity, portfolioAmount);
        }

        Transaction transaction = new Transaction(
                portfolio,
                asset,
                type,
                quantity,
                price,
                total,
                asset.getCurrency(),
                executionFxRate,
                portfolioAmount
        );

        transactionRepository.saveAndFlush(transaction);

        notificationOutbox.recordTrade(
                portfolio.getId(),
                transaction.getId(),
                asset.getSymbol(),
                type,
                quantity
        );

        eventPublisher.publishEvent(new PortfolioTradeCommitted(portfolio.getId()));

        return new OrderResponse(
                transaction.getId(),
                portfolio.getId(),
                asset.getSymbol(),
                type,
                quantity,
                price,
                total,
                asset.getCurrency(),
                executionFxRate,
                portfolioAmount,
                portfolio.getCurrency(),
                portfolio.getCashBalance()
        );
    }

    private void processBuy(
            Portfolio portfolio,
            Asset asset,
            Holding holding,
            BigDecimal quantity,
            BigDecimal price,
            BigDecimal total
    ) {
        if (portfolio.getCashBalance().compareTo(total) < 0) {
            throw new ResponseStatusException(
                    HttpStatus.BAD_REQUEST,
                    "Insufficient virtual cash"
            );
        }

        if (holding == null) {
            Holding created = new Holding(
                    portfolio, asset, quantity, price
            );
            created.setHistoricalCostPortfolio(total);
            holdingRepository.save(created);
        } else {
            BigDecimal oldQuantity = holding.getQuantity();
            BigDecimal newQuantity = oldQuantity.add(quantity);

            BigDecimal oldCost = oldQuantity.multiply(
                    holding.getAverageBuyPrice()
            );

            BigDecimal newCost = quantity.multiply(price);

            BigDecimal newAverage = oldCost.add(newCost)
                    .divide(newQuantity, 4, RoundingMode.HALF_UP);

            BigDecimal updatedHistoricalCost = holding
                    .getHistoricalCostPortfolio()
                    .add(total);

            holding.updatePosition(newQuantity, newAverage);
            holding.setHistoricalCostPortfolio(updatedHistoricalCost);
        }

        portfolio.setCashBalance(
                portfolio.getCashBalance().subtract(total)
        );
    }

    private void processSell(
            Portfolio portfolio,
            Holding holding,
            BigDecimal quantity,
            BigDecimal total
    ) {
        if (holding == null ||
                holding.getQuantity().compareTo(quantity) < 0) {
            throw new ResponseStatusException(
                    HttpStatus.BAD_REQUEST,
                    "Insufficient holdings"
            );
        }

        BigDecimal originalQuantity = holding.getQuantity();
        BigDecimal remaining = originalQuantity.subtract(quantity);

        if (remaining.signum() == 0) {
            holdingRepository.delete(holding);
        } else {
            BigDecimal remainingHistoricalCost = holding
                    .getHistoricalCostPortfolio()
                    .multiply(remaining)
                    .divide(originalQuantity, 4, RoundingMode.HALF_UP);

            holding.updatePosition(
                    remaining,
                    holding.getAverageBuyPrice()
            );

            holding.setHistoricalCostPortfolio(
                    remainingHistoricalCost
            );
        }

        portfolio.setCashBalance(
                portfolio.getCashBalance().add(total)
        );
    }

	mux.HandleFunc("GET /health", func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("Content-Type", "application/json")

		json.NewEncoder(w).Encode(map[string]string{
			"status":  "UP",
			"service": "market-simulator",
		})
	})

	mux.HandleFunc("GET /prices", func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("Content-Type", "application/json")
		json.NewEncoder(w).Encode(simulator.GetPrices())
	})

	// WebSocket endpoint for live simulated prices.
	mux.HandleFunc("GET /ws/prices", func(w http.ResponseWriter, r *http.Request) {
		conn, err := websocket.Accept(w, r, &websocket.AcceptOptions{
			OriginPatterns: []string{"localhost:*", "127.0.0.1:*"},
		})
		if err != nil {
			log.Printf("WebSocket connection rejected: %v", err)
			return
		}
		defer conn.CloseNow()

		ctx := r.Context()

		sendPrices := func() error {
			prices := simulator.GetPrices()

			data, err := json.Marshal(prices)
			if err != nil {
				return err
			}

			return conn.Write(ctx, websocket.MessageText, data)
		}

		// Send prices immediately after connecting.
		if err := sendPrices(); err != nil {
			return
		}

		ticker := time.NewTicker(5 * time.Second)
		defer ticker.Stop()

		for {
			select {
			case <-ctx.Done():
				return
			case <-ticker.C:
				if err := sendPrices(); err != nil {
					return
				}
			}
		}
	})

	// One shared market simulation for all connected clients.
	go func() {
		ticker := time.NewTicker(5 * time.Second)
		defer ticker.Stop()

		for range ticker.C {
			simulator.UpdatePrices()
		}
	}()

	log.Println("Market simulator listening on http://localhost:8081")
	log.Println("WebSocket endpoint: ws://localhost:8081/ws/prices")

	if err := http.ListenAndServe("127.0.0.1:8081", mux); err != nil {
		log.Fatal(err)
	}
}

func (m *MarketSimulator) GetPrices() []AssetPrice {
 m.mu.RLock()
 defer m.mu.RUnlock()
 result := make([]AssetPrice, 0, len(m.prices))
 for _, price := range m.prices { result = append(result, price) }
 return result
}

func (m *MarketSimulator) UpdatePrices() {
 m.mu.Lock()
 defer m.mu.Unlock()
 now := time.Now().UTC()
 for symbol, asset := range m.prices {
  change := (rand.Float64() - 0.5) * 2 * m.volatility[symbol]
  asset.Price = asset.Price * (1 + change)
  asset.UpdatedAt = now
  m.prices[symbol] = asset
 }
}

	simulator.UpdatePrices()

	prices := simulator.GetPrices()
