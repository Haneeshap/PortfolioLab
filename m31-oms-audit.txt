package com.portfoliolab.backend.oms;

import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Service;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;
import java.util.List;

@Service
class ReconciliationService {
    private final JdbcTemplate jdbc;
    ReconciliationService(JdbcTemplate jdbc) { this.jdbc = jdbc; }

    record ExceptionItem(long orderId, String code, String details) {}
    record Report(long portfolioId, int checkedOrders, int exceptionCount, List<ExceptionItem> exceptions,
                  String scope) {}

    Report reconcile(long portfolioId) {
        Integer exists = jdbc.queryForObject("SELECT count(*) FROM portfolios WHERE id=?", Integer.class, portfolioId);
        if (exists == null || exists == 0) throw new org.springframework.web.server.ResponseStatusException(
                org.springframework.http.HttpStatus.NOT_FOUND, "Portfolio not found");
        Integer count = jdbc.queryForObject("SELECT count(*) FROM simulated_orders WHERE portfolio_id=?", Integer.class, portfolioId);
        List<ExceptionItem> issues = jdbc.query("""
            WITH receipts AS (
              SELECT order_id, COUNT(*) AS receipt_count, COALESCE(SUM(quantity),0) AS receipt_qty
              FROM simulated_exchange_executions GROUP BY order_id
            ), fills AS (
              SELECT order_id, COUNT(*) AS fill_count, COALESCE(SUM(quantity),0) AS fill_qty
              FROM simulated_order_fills GROUP BY order_id
            ), linked AS (
              SELECT e.order_id,
                     COUNT(*) FILTER (WHERE t.id IS NULL OR f.id IS NULL) AS missing_links,
                     COUNT(*) FILTER (WHERE t.id IS NOT NULL AND (
                         t.portfolio_id <> o.portfolio_id OR t.asset_id <> o.asset_id OR
                         t.transaction_type <> o.side OR t.quantity <> e.quantity OR
                         t.price_per_unit <> e.price_per_unit OR f.quantity <> e.quantity OR
                         f.order_id <> e.order_id OR f.fill_sequence <> e.fill_sequence OR
                         f.transaction_id <> e.transaction_id
                     )) AS mismatched_trades
              FROM simulated_exchange_executions e
              JOIN simulated_orders o ON o.id=e.order_id
              LEFT JOIN transactions t ON t.id=e.transaction_id
              LEFT JOIN simulated_order_fills f ON f.transaction_id=e.transaction_id
              GROUP BY e.order_id
            )
            SELECT o.id,
              CASE
                WHEN COALESCE(l.missing_links,0)>0 THEN 'MISSING_SETTLEMENT_LINK'
                WHEN COALESCE(l.mismatched_trades,0)>0 THEN 'EXECUTION_TRANSACTION_MISMATCH'
                WHEN COALESCE(r.receipt_count,0)<>COALESCE(f.fill_count,0) THEN 'RECEIPT_FILL_COUNT_MISMATCH'
                WHEN COALESCE(r.receipt_qty,0)<>COALESCE(f.fill_qty,0) THEN 'RECEIPT_FILL_QUANTITY_MISMATCH'
                WHEN o.filled_quantity<>COALESCE(r.receipt_qty,0) THEN 'ORDER_FILLED_QUANTITY_MISMATCH'
                WHEN o.status='FILLED' AND o.filled_quantity<>o.requested_quantity THEN 'INVALID_FILLED_STATUS'
                WHEN o.status='PARTIALLY_FILLED' AND (o.filled_quantity<=0 OR o.filled_quantity>=o.requested_quantity) THEN 'INVALID_PARTIAL_STATUS'
                WHEN o.status IN ('SUBMITTED','ACCEPTED','REJECTED') AND o.filled_quantity<>0 THEN 'INVALID_UNFILLED_STATUS'
                WHEN o.status='CANCELLED' AND o.filled_quantity>o.requested_quantity THEN 'INVALID_CANCELLED_STATUS'
                ELSE NULL END AS code
            FROM simulated_orders o
            LEFT JOIN receipts r ON r.order_id=o.id
            LEFT JOIN fills f ON f.order_id=o.id
            LEFT JOIN linked l ON l.order_id=o.id
            WHERE o.portfolio_id=?
            ORDER BY o.id
            """, (rs,n)->new ExceptionItem(rs.getLong("id"),rs.getString("code"),
                    "Inconsistent simulated order, execution receipt, or settled trade"), portfolioId)
                .stream().filter(e -> e.code()!=null).toList();
        return new Report(portfolioId, count == null ? 0 : count, issues.size(), issues,
                "Internal consistency only; receipts and settlements originate in the same Spring Boot worker");
    }
}

@RestController
@RequestMapping("/api/portfolios/{portfolioId}/reconciliation")
class ReconciliationController {
    private final ReconciliationService service;
    ReconciliationController(ReconciliationService service) { this.service=service; }
    @GetMapping
    ReconciliationService.Report report(@PathVariable long portfolioId) {
        return service.reconcile(portfolioId);
    }
}

package com.portfoliolab.backend.oms;

import com.portfoliolab.backend.portfolio.OrderRequest;
import com.portfoliolab.backend.portfolio.OrderResponse;
import com.portfoliolab.backend.portfolio.OrderService;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;
import org.springframework.transaction.PlatformTransactionManager;
import org.springframework.transaction.support.TransactionTemplate;
import org.springframework.web.server.ResponseStatusException;
import java.math.BigDecimal;
import java.util.List;
import java.util.UUID;

/** Local exchange execution simulator. Uses the existing settlement service atomically.
 * Phase 3 can replace the local adapter with an independent Go exchange service.
 */
@Component
public class SimulatedExchangeWorker {
    private static final Logger log = LoggerFactory.getLogger(SimulatedExchangeWorker.class);
    private final JdbcTemplate jdbc;
    private final OrderService settlement;
    private final TransactionTemplate tx;

    public SimulatedExchangeWorker(JdbcTemplate jdbc, OrderService settlement, PlatformTransactionManager manager) {
        this.jdbc = jdbc;
        this.settlement = settlement;
        this.tx = new TransactionTemplate(manager);
    }

    record Pending(long id, long portfolioId, long assetId, String side,
                   BigDecimal requested, BigDecimal filled, String status) {}

    @Scheduled(fixedDelay = 2500, initialDelay = 5000)
    public void executeNext() {
        final long[] attemptedId = {-1};
        // Each cycle processes at most one order. Lock prevents concurrent fills/cancellation.
        try {
            tx.executeWithoutResult(ignored -> {
                List<Pending> pending = jdbc.query("""
                    SELECT id, portfolio_id, asset_id, side, requested_quantity,
                           filled_quantity, status
                    FROM simulated_orders
                    WHERE status IN ('SUBMITTED','ACCEPTED','PARTIALLY_FILLED')
                    ORDER BY id ASC LIMIT 1 FOR UPDATE SKIP LOCKED
                    """, (rs, row) -> new Pending(
                        rs.getLong("id"), rs.getLong("portfolio_id"), rs.getLong("asset_id"),
                        rs.getString("side"), rs.getBigDecimal("requested_quantity"),
                        rs.getBigDecimal("filled_quantity"), rs.getString("status")));
                if (pending.isEmpty()) return;
                Pending order = pending.get(0);
                attemptedId[0] = order.id();
                if (order.status().equals("SUBMITTED")) {
                    jdbc.update("UPDATE simulated_orders SET status='ACCEPTED',updated_at=NOW() WHERE id=?", order.id());
                    return;
                }
                BigDecimal remaining = order.requested().subtract(order.filled());
                if (remaining.signum() <= 0) throw new IllegalStateException("Invalid pending order remaining quantity");
                // Integer share quantities: fill half (rounded up), then remainder next cycle.
                BigDecimal fillQty = remaining.compareTo(BigDecimal.ONE) <= 0
                    ? remaining : remaining.divide(new BigDecimal("2"), 0, java.math.RoundingMode.CEILING);
                // Settlement enforces risk, buying power, holdings, pricing, FX, notifications.
                // Its @Transactional method joins this transaction; rollback is all-or-nothing.
                OrderResponse trade = settlement.placeOrder(order.portfolioId(),
                    new OrderRequest(order.assetId(), order.side(), fillQty));
                int sequence = jdbc.queryForObject("""
                    SELECT COALESCE(MAX(fill_sequence),0)+1
                    FROM simulated_exchange_executions WHERE order_id=?
                    """, Integer.class, order.id());
                String reference = "LOCAL-" + UUID.randomUUID();
                jdbc.update("""
                    INSERT INTO simulated_exchange_executions
                    (order_id,execution_ref,fill_sequence,transaction_id,quantity,price_per_unit)
                    VALUES (?,?,?,?,?,?)
                    """, order.id(), reference, sequence, trade.transactionId(), fillQty, trade.pricePerUnit());
                jdbc.update("""
                    INSERT INTO simulated_order_fills(order_id,transaction_id,fill_sequence,quantity)
                    VALUES (?,?,?,?)
                    """, order.id(), trade.transactionId(), sequence, fillQty);
                BigDecimal cumulative = order.filled().add(fillQty);
                String status = cumulative.compareTo(order.requested()) == 0 ? "FILLED" : "PARTIALLY_FILLED";
                jdbc.update("""
                    UPDATE simulated_orders SET filled_quantity=?,status=?,updated_at=NOW() WHERE id=?
                    """, cumulative, status, order.id());
                log.info("Simulated order {}: {} {} shares; {}", order.id(), reference, fillQty, status);
            });
        } catch (ResponseStatusException ex) {
            if (ex.getStatusCode().is5xxServerError()) {
                log.warn("Simulated execution temporarily unavailable; will retry: {}", ex.getReason());
                return;
            }
            // A permanent validation/risk failure rejects the order; no settlement was committed.
            String reason = ex.getReason() == null ? "Order rejected by settlement" : ex.getReason();
            try {
                if (attemptedId[0] < 0) return;
                tx.executeWithoutResult(ignored -> jdbc.update("""
                    UPDATE simulated_orders SET status='REJECTED',failure_reason=?,updated_at=NOW()
                    WHERE id=? AND status IN ('SUBMITTED','ACCEPTED','PARTIALLY_FILLED')
                    """, reason, attemptedId[0]));
            } catch (Exception failure) { log.warn("Unable to persist rejection", failure); }
            log.warn("Simulated execution rejected: {}", reason);
        } catch (Exception ex) {
            // Transient market/FX or infrastructure failures must not silently reject orders.
            log.warn("Simulated exchange cycle rolled back; will retry: {}", ex.toString());
        }
    }
}

package com.portfoliolab.backend.oms;

import com.portfoliolab.backend.portfolio.OrderRequest;
import org.springframework.http.HttpStatus;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.server.ResponseStatusException;
import java.math.BigDecimal;
import java.util.List;
import java.util.Locale;

@Service
public class SimulatedOrderService {
 private final JdbcTemplate jdbc;
 public SimulatedOrderService(JdbcTemplate jdbc) { this.jdbc=jdbc; }
 public record OrderView(long id,long portfolioId,long assetId,String side,BigDecimal requestedQuantity,
                         BigDecimal filledQuantity,String status,String failureReason) {}
 private final org.springframework.jdbc.core.RowMapper<OrderView> row=(rs,n)->new OrderView(
  rs.getLong("id"),rs.getLong("portfolio_id"),rs.getLong("asset_id"),rs.getString("side"),
  rs.getBigDecimal("requested_quantity"),rs.getBigDecimal("filled_quantity"),
  rs.getString("status"),rs.getString("failure_reason"));
 @Transactional
 public OrderView submit(long portfolioId, OrderRequest request) {
  if(request==null || request.assetId()==null || request.type()==null || request.quantity()==null)
   throw new ResponseStatusException(HttpStatus.BAD_REQUEST,"Asset, side and quantity required");
  String side=request.type().trim().toUpperCase(Locale.ROOT);
  BigDecimal qty=request.quantity();
  if(!List.of("BUY","SELL").contains(side) || qty.signum()<=0 || qty.stripTrailingZeros().scale()>0 || qty.precision()>13)
   throw new ResponseStatusException(HttpStatus.BAD_REQUEST,"Invalid side or quantity");
  Integer p=jdbc.queryForObject("SELECT count(*) FROM portfolios WHERE id=?",Integer.class,portfolioId);
  Integer a=jdbc.queryForObject("SELECT count(*) FROM assets WHERE id=?",Integer.class,request.assetId());
  if(p==null||p==0||a==null||a==0) throw new ResponseStatusException(HttpStatus.NOT_FOUND,"Portfolio or asset not found");
  Long id=jdbc.queryForObject("INSERT INTO simulated_orders(portfolio_id,asset_id,side,requested_quantity) VALUES(?,?,?,?) RETURNING id",
   Long.class,portfolioId,request.assetId(),side,qty);
  return get(portfolioId,id);
 }
 public OrderView get(long portfolioId,long orderId){
  return jdbc.query("SELECT * FROM simulated_orders WHERE id=? AND portfolio_id=?",row,orderId,portfolioId)
   .stream().findFirst().orElseThrow(()->new ResponseStatusException(HttpStatus.NOT_FOUND,"Order not found"));
 }
 public List<OrderView> list(long portfolioId){
  return jdbc.query("SELECT * FROM simulated_orders WHERE portfolio_id=? ORDER BY id DESC LIMIT 100",row,portfolioId);
 }
 @Transactional
 public OrderView cancel(long portfolioId,long orderId){
  List<OrderView> locked=jdbc.query("SELECT * FROM simulated_orders WHERE id=? AND portfolio_id=? FOR UPDATE",row,orderId,portfolioId);
  if(locked.isEmpty()) throw new ResponseStatusException(HttpStatus.NOT_FOUND,"Order not found");
  OrderView current=locked.get(0);
  if(!List.of("SUBMITTED","ACCEPTED","PARTIALLY_FILLED").contains(current.status()))
   throw new ResponseStatusException(HttpStatus.CONFLICT,"Order is already terminal");
  jdbc.update("UPDATE simulated_orders SET status='CANCELLED',updated_at=NOW() WHERE id=?",orderId);
  return get(portfolioId,orderId);
 }

}
